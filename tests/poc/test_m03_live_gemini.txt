            ],
            tools=[_WEATHER_TOOL],
            tool_choice="auto",
            actor="front",
            scenario="user_input",
            max_tokens=512,
            temperature=0.0,
        )
        response = await adapter.generate(request)

        assert "city" in tc.arguments, f"Expected city arg: {tc.arguments}"
        assert tc.id.startswith("call_"), f"Expected synthetic ID: {tc.id}"
        assert response.finish_reason == "tool_calls"

    @pytest.mark.asyncio
    async def test_tool_call_with_result_round_trip(self, adapter: GeminiConciergeAdapter) -> None:
        """Full round-trip: tool call -> tool result -> final answer."""
        # Step 1: Initial request triggers tool call
        request1 = ConciergeModelRequest(
            capability="TOOL_CALL",
            system_prompt=(
                "You have access to a weather tool. Always use it when asked about weather. "
                "After receiving the tool result, provide a natural language summary."
            ),
            messages=[
                ModelMessage(role="user", content="What is the weather in Tokyo?"),
            ],
            tools=[_WEATHER_TOOL],
            tool_choice="auto",
            actor="front",
            scenario="user_input",
            max_tokens=512,
            temperature=0.0,
        )
        response1 = await adapter.generate(request1)

        assert response1.has_tool_calls, f"Step 1 should produce tool call: {response1!r}"
        tc = response1.tool_calls[0]

        # Step 2: Provide tool result, get final answer
        request2 = ConciergeModelRequest(
            capability="TOOL_CALL",
            system_prompt=request1.system_prompt,
            messages=[
                ModelMessage(role="user", content="What is the weather in Tokyo?"),
                ModelMessage(
                    role="assistant",
                    content=response1.text,
                    tool_calls=response1.tool_calls,
                ),
                ModelMessage(
                    role="tool",
                    content=json.dumps({"temp_c": 22, "condition": "partly cloudy"}),
                    tool_call_id=tc.id,
                    name=tc.name,
                ),
            ],
            tools=[_WEATHER_TOOL],
            tool_choice="auto",
            actor="front",
            scenario="user_input",
            max_tokens=512,
            temperature=0.0,
        )
        response2 = await adapter.generate(request2)

        assert response2.has_text, f"Step 2 should produce text: {response2!r}"
        text_lower = response2.text.lower()
        assert (
            "tokyo" in text_lower or "22" in text_lower or "cloud" in text_lower
        ), f"Final answer should reference tool result: {response2.text}"
        assert response2.finish_reason == "stop"

# 3. Streaming (generate_stream)

class TestStreaming:
    """Verify streaming text and tool calls via generate_stream()."""

    @pytest.mark.asyncio
    async def test_stream_text_chunks(self, adapter: GeminiConciergeAdapter) -> None:
        """Streaming yields text_delta chunks and ends with done."""
        request = ConciergeModelRequest(
            capability="CHAT",
            system_prompt="You are a helpful assistant. Write a short paragraph about the moon.",
            messages=[
                ModelMessage(role="user", content="Tell me about the moon in 2-3 sentences."),
            ],
            actor="front",
            scenario="user_input",
            max_tokens=256,
            temperature=0.3,
        )

        chunks: list[StreamChunk] = []
        async for chunk in adapter.generate_stream(request):
            chunks.append(chunk)

        assert len(chunks) >= 2, f"Expected multiple chunks, got {len(chunks)}"

        # Last chunk should be 'done'
        done_chunks = [c for c in chunks if c.chunk_type == "done"]
        assert len(done_chunks) == 1, f"Expected exactly one done chunk, got {len(done_chunks)}"
        done = done_chunks[0]
        assert done.response is not None
        assert isinstance(done.response, ConciergeModelResponse)
        assert done.response.model_id, "done chunk should have model_id"
        assert done.response.latency_ms > 0
        assert done.response.finish_reason in ("stop", "tool_calls")

        # Text deltas should assemble into full text
        text_chunks = [c for c in chunks if c.chunk_type == "text_delta"]
        if text_chunks:
            assembled_text = "".join(c.text for c in text_chunks)
            assert len(assembled_text) > 10, f"Expected substantial text: {assembled_text}"
            # The done response text should match assembled deltas
            assert (
                done.response.text == assembled_text
            ), f"Done text mismatch.\nAssembled: {assembled_text}\nDone: {done.response.text}"

    @pytest.mark.asyncio
    async def test_stream_tool_call(self, adapter: GeminiConciergeAdapter) -> None:
        """Streaming with tool calls yields tool_call_delta chunks."""
        request = ConciergeModelRequest(
            capability="TOOL_CALL",
            system_prompt="You have tools. Use them when needed. Do not answer without a tool.",
            messages=[
                ModelMessage(role="user", content="What is the weather in Berlin?"),
            ],
            tools=[_WEATHER_TOOL],
            tool_choice="auto",
            actor="front",
            scenario="user_input",
            max_tokens=256,
            temperature=0.0,
        )

        chunks: list[StreamChunk] = []
        async for chunk in adapter.generate_stream(request):
            chunks.append(chunk)

        done_chunks = [c for c in chunks if c.chunk_type == "done"]
        assert len(done_chunks) == 1
        done = done_chunks[0]

        assert done.response is not None
        # Streaming tool calls: either in done response or via tool_call_delta
        tool_deltas = [c for c in chunks if c.chunk_type == "tool_call_delta"]

        if done.response.has_tool_calls:
            tc = done.response.tool_calls[0]
            assert tc.name == "get_weather"
            assert "city" in tc.arguments
        elif tool_deltas:
            # Tool call appeared via delta chunks
            assert any("get_weather" in str(c.tool_call_partial) for c in tool_deltas)
        else:
            # Model may have answered directly -- at least verify we got a response
            assert done.response.has_text, f"Expected text or tool call: {done.response!r}"

class TestStructuredOutput:
    """Verify the adapter can produce structured JSON output."""

    @pytest.mark.asyncio
    async def test_json_output(self, adapter: GeminiConciergeAdapter) -> None:
        """Model produces valid JSON when asked."""
        request = ConciergeModelRequest(
            capability="STRUCTURED",
            system_prompt=(
                "You are a JSON generator. Always respond with valid JSON only, no markdown, no explanation. "
                'Output format: {"name": string, "age": number, "city": string}'
            ),
            messages=[
                ModelMessage(
                    role="user",
                    content="Generate a JSON object for a person named Alice who is 30 years old and lives in London.",
                ),
            ],
            actor="back",
            scenario="structured_output",
            max_tokens=256,
            temperature=0.0,
        )
        response = await adapter.generate(request)

        assert response.has_text
        # Strip markdown code fences if present
        text = response.text.strip()
        if text.startswith("```"):
            lines = text.split("\n")
            lines = [line for line in lines if not line.strip().startswith("```")]
            text = "\n".join(lines).strip()

        parsed = json.loads(text)
        assert isinstance(parsed, dict), f"Expected dict, got: {type(parsed)}"
        assert "name" in parsed
        assert "age" in parsed
        assert "city" in parsed
        assert parsed["name"].lower() == "alice"
        assert parsed["age"] == 30
        assert parsed["city"].lower() == "london"

    @pytest.mark.asyncio
    async def test_json_list_output(self, adapter: GeminiConciergeAdapter) -> None:
        """Model produces a JSON array."""
        request = ConciergeModelRequest(
            capability="STRUCTURED",
            system_prompt=(
                "You are a JSON generator. Always respond with a valid JSON array only, "
                'no markdown fences, no explanation. Each item has: {"item": string, "count": number}'
            ),
            messages=[
                ModelMessage(
                    role="user",
                    content="List 3 fruits with their counts: apple 5, banana 3, cherry 8.",
                ),
            ],
            actor="back",
            scenario="structured_output",
            max_tokens=256,
            temperature=0.0,
        )
        response = await adapter.generate(request)

        assert response.has_text
        text = response.text.strip()
        if text.startswith("```"):
            lines = text.split("\n")
            lines = [line for line in lines if not line.strip().startswith("```")]
            text = "\n".join(lines).strip()

        parsed = json.loads(text)
        assert isinstance(parsed, list), f"Expected list, got: {type(parsed)}"
        assert len(parsed) == 3
        names = {item["item"].lower() for item in parsed}
        assert "apple" in names
        assert "banana" in names
        assert "cherry" in names

    @pytest.mark.asyncio
    async def test_multiple_consecutive_tool_results(self, adapter: GeminiConciergeAdapter) -> None:
        """Multiple tool results in sequence are merged correctly (Gemini requirement)."""
        request = ConciergeModelRequest(
            capability="TOOL_CALL",
            system_prompt=(
                "You have tools. After receiving all tool results, "
                "summarize the information in one sentence."
            ),
            messages=[
                ModelMessage(role="user", content="What is the weather in Tokyo and Berlin?"),
                ModelMessage(
                    role="assistant",
                    content="",
                    tool_calls=[
                        ToolCallResult(
                            id="call_001", name="get_weather", arguments={"city": "Tokyo"}
                        ),
                        ToolCallResult(
                            id="call_002", name="get_weather", arguments={"city": "Berlin"}
                        ),
                    ],
                ),
                # Two consecutive tool results -- adapter must merge into single Content
                ModelMessage(
                    role="tool",
                    content=json.dumps({"temp_c": 25, "condition": "sunny"}),
                    tool_call_id="call_001",
                    name="get_weather",
                ),
                ModelMessage(
                    role="tool",
                    content=json.dumps({"temp_c": 12, "condition": "rainy"}),
                    tool_call_id="call_002",
                    name="get_weather",
                ),
            ],
            tools=[_WEATHER_TOOL],
            tool_choice="auto",
            actor="front",
            scenario="user_input",
            max_tokens=256,
            temperature=0.0,
        )
        response = await adapter.generate(request)

        assert response.has_text, f"Expected text summary: {response!r}"
        text_lower = response.text.lower()
        # Should reference both cities or their weather data
        has_context = ("tokyo" in text_lower or "25" in text_lower or "sunny" in text_lower) and (
            "berlin" in text_lower or "12" in text_lower or "rain" in text_lower
        )
        assert has_context, f"Expected both cities in response: {response.text}"

class TestConcurrency:
    """Verify adapter handles concurrent requests correctly."""

    @pytest.mark.asyncio
    async def test_parallel_requests(self, adapter: GeminiConciergeAdapter) -> None:
        """Multiple concurrent requests all succeed."""
        requests = [
            ConciergeModelRequest(
                capability="CHAT",
                system_prompt="Reply with one word only.",
                messages=[ModelMessage(role="user", content=f"Say the number {i}.")],
                actor="front",
                scenario="ack",
                max_tokens=256,
                temperature=0.0,
            )
            for i in range(3)
        ]

        responses = await asyncio.gather(*[adapter.generate(r) for r in requests])

        assert len(responses) == 3
        for resp in responses:
            assert isinstance(resp, ConciergeModelResponse)
            assert resp.has_text
            assert resp.model_id
